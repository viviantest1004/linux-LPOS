# config.txt - VideoCore start.elf 가 읽는 부팅 설정.
# 이 파일은 ARM 이 아니라 GPU 가 파싱한다. ARM 코어를 깨우기 전에 처리된다.

# ── 필수 ────────────────────────────────────────────────────────
# 64비트 모드. 이걸 켜야 start.elf 가 64비트 커널을 찾고,
# ARM 코어를 AArch64 EL2 로 진입시킨다.
arm_64bit=1

# 로드할 이미지 파일명.
# arm_64bit=1 의 기본값은 kernel8.img 지만 kernel= 로 바꿀 수 있다.
# 반드시 config.mk 의 KERNEL_IMAGE 와 같아야 한다.
kernel=test_a_123_LPzero2W.img

# 커널 로드 주소. 우리 linker.ld 의 0x80000 과 반드시 일치해야 한다.
kernel_address=0x80000

# ── 시리얼 콘솔 ─────────────────────────────────────────────────
# UART 를 활성화하고 클럭을 48MHz 로 고정한다.
# 고정하지 않으면 펌웨어 버전/온도에 따라 클럭이 흔들려 보레이트가 깨진다.
enable_uart=1
init_uart_clock=48000000
init_uart_baud=115200

# ── 부팅 시간 단축 ──────────────────────────────────────────────
# 무지개 스플래시 화면 생략
disable_splash=1
# 기본 1초 대기 제거
boot_delay=0
# 부팅 로고 끄기
disable_overscan=1

# ── 메모리 분할 ─────────────────────────────────────────────────
# Zero 2 W 는 RAM 이 512MB 뿐이라 GPU 몫이 곧 ARM 의 손실이다.
#   16  : 완전 헤드리스 (프레임버퍼 없음) - 최소값
#   64  : 프레임버퍼 사용 (Phase 2 에서 필요)
#   128 : 하드웨어 비디오 디코딩
gpu_mem=64

# ── HDMI ────────────────────────────────────────────────────────
# 화면이 검게 나오면 아래 주석을 푼다.
# 펌웨어는 부팅 시점에 모니터가 붙어 있는지(핫플러그 신호)를 보고
# HDMI 출력을 켤지 정한다. 모니터를 Pi 보다 늦게 켜거나, 일부
# 모니터/캡처보드/KVM 은 이 신호를 제대로 주지 않아 출력이 꺼진 채로
# 부팅된다. hdmi_force_hotplug=1 은 감지 결과와 무관하게 출력을 켠다.
#hdmi_force_hotplug=1

# 감지가 안 될 때 쓸 기본 모드. hdmi_force_hotplug 와 같이 쓴다.
# group=1(CEA) mode=16 = 1080p60,  group=2(DMT) mode=16 = 1024x768@60
#hdmi_group=1
#hdmi_mode=16

# ── 전원 절약 ───────────────────────────────────────────────────
# HDMI 출력을 안 쓸 때 켜면 약 25mA 절약. 화면이 필요하면 주석 처리.
#hdmi_blanking=2

# 온보드 ACT LED 를 펌웨어가 건드리지 않게 (우리가 직접 제어)
dtparam=act_led_trigger=none
